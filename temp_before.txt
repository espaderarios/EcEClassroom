function toKebabCase(value) {
  return String(value)
    .replace(/([a-z0-9])([A-Z])/g, "$1-$2")
    .replace(/[_\s]+/g, "-")
    .toLowerCase();
}

function toTitleCase(value) {
  return String(value)
    .replace(/([a-z0-9])([A-Z])/g, "$1 $2")
    .replace(/[-_]+/g, " ")
    .replace(/\s+/g, " ")
    .trim()
    .replace(/\b\w/g, (char) => char.toUpperCase());
}

function ensureIconDefinition(key) {
  if (!key) {
    return null;
  }

  if (!ICON_LIBRARY[key]) {
    const fileName = toKebabCase(key);
    ICON_LIBRARY[key] = {
      file: `icons/${fileName}.svg`,
      alt: `${toTitleCase(key) || "Interface"} icon`
    };
  }

  return ICON_LIBRARY[key];
}

function resolveIconKey(raw) {
  if (!raw) return null;
  if (ICON_LIBRARY.hasOwnProperty(raw)) return raw;
  if (LEGACY_ICON_MAP.hasOwnProperty(raw)) return LEGACY_ICON_MAP[raw];
  return raw;
}

function getIconDefinition(raw) {
  const key = resolveIconKey(raw);
  if (key) {
    const definition = ICON_LIBRARY[key] || ensureIconDefinition(key);
    if (definition) {
      return definition;
    }
  }
  return ICON_LIBRARY.default || null;
}

function escapeAttribute(value) {
  if (value === null || value === undefined) return "";
  return String(value)
    .replace(/&/g, "&amp;")
    .replace(/"/g, "&quot;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

function escapeHtml(value) {
  if (value === null || value === undefined) return "";
  return String(value)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}

function tryParseLooseJsonString(value) {
  if (!value || typeof value !== 'string') return null;
  let trimmed = value.trim();
  if (!trimmed) return null;
  trimmed = trimmed.replace(/^Answer:\s*/i, '').replace(/^Question:\s*/i, '').trim();
  if (!trimmed) return null;
  if (!/^[{\[]/.test(trimmed)) return null;
  const candidate = trimmed.replace(/,+$/, '');
  try {
    return JSON.parse(candidate);
  } catch (err) {
    return null;
  }
}

function decodeJsonFragment(value) {
  if (typeof value !== 'string') return value;
  try {
    const safeValue = value
      .replace(/\\/g, '\\\\')
      .replace(/"/g, '\\"')
      .replace(/\r/g, '\\r')
      .replace(/\n/g, '\\n')
      .replace(/\t/g, '\\t');
    return JSON.parse(`"${safeValue}"`);
  } catch (err) {
    return value
      .replace(/\\n/g, '\n')
      .replace(/\\t/g, '\t')
      .replace(/\\r/g, '\r')
      .replace(/\\"/g, '"')
      .replace(/\\\\/g, '\\');
  }
}

function normalizeCardText(raw, preferredKeys = [], fallback = '') {
  if (raw == null) {
    return fallback;
  }

  if (typeof raw !== 'string') {
    return String(raw);
  }

  let text = raw.trim();
  if (!text) return fallback;

  text = text.replace(/^Answer:\s*/i, '').replace(/^Question:\s*/i, '').trim();
  if (!text) return fallback;

  const parsed = tryParseLooseJsonString(text);
  if (parsed) {
    const container = Array.isArray(parsed) ? parsed[0] : parsed;
    if (container && typeof container === 'object') {
      for (const key of preferredKeys) {
        const candidate = container[key];
        if (typeof candidate === 'string' && candidate.trim()) {
          return candidate.trim();
        }
      }
    }
  }

  const loweredKeys = preferredKeys.map(key => key.toLowerCase());
  const fieldPattern = /"(question|answer|front|back|prompt|response|explanation)"\s*:\s*"([^\"]*)"/gi;
  let match;
  while ((match = fieldPattern.exec(text))) {
    const key = match[1].toLowerCase();
    const value = decodeJsonFragment(match[2]).trim();
    if (!value) continue;
    if (loweredKeys.length === 0 || loweredKeys.includes(key)) {
      return value;
    }
  }

  return text;
}

function getCardDisplayText(card, index = null) {
  const position = typeof index === 'number' && index >= 0 ? index + 1 : null;
  const questionFallback = position ? `Question ${position}` : '';
  const answerFallback = position ? `Answer ${position}` : '';

  const rawQuestion = normalizeCardText(card?.question, ['question', 'front', 'prompt'], '');
  const question = (rawQuestion && rawQuestion.trim()) || questionFallback || '';

  const rawAnswer = normalizeCardText(card?.answer, ['answer', 'back', 'response', 'explanation'], '');
  const baseAnswer = rawAnswer && rawAnswer.trim() ? rawAnswer : '';
  const defaultAnswer = answerFallback || 'No answer provided yet.';

  let answer = baseAnswer || defaultAnswer;
  if (answer && question && answer.trim().toLowerCase() === question.trim().toLowerCase()) {
    answer = defaultAnswer;
  }

  return {
    question,
    answer
  };
}

function renderIcon(raw, altText, size = 32, className = "") {
  const iconDef = getIconDefinition(raw);
  if (!iconDef) {
    return altText ? `<span class="sr-only">${altText}</span>` : "";
  }

  const widthAttr = size ? ` width="${size}" height="${size}"` : "";
  const responsiveClassName = ["app-icon", className].filter(Boolean).join(" ");
  const classAttr = ` class="${escapeAttribute(responsiveClassName)}"`;
  const styleAttr = size ? ` style="--icon-size:${size}px;"` : "";
  const alt = escapeAttribute(altText || iconDef.alt || "Icon");
  const src = escapeAttribute(iconDef.file);

  return `<img src="${src}" alt="${alt}" loading="lazy"${widthAttr}${classAttr}${styleAttr}>`;
}

const LEGACY_ICON_PATTERN = /≡ƒ[^\s<"']+|Γ(?!Ç)[^\s<"']+/g;

function renderLegacyIcon(raw, altText, size = 20, className = "inline-icon") {
  const key = resolveIconKey(raw) || raw;
  const alt = altText || LEGACY_ICON_ALTS[raw] || (ICON_LIBRARY[key]?.alt) || "Interface icon";
  return renderIcon(key, alt, size, className);
}

function replaceLegacyIcons(html) {
  if (
    typeof html !== "string" ||
    (html.indexOf("≡ƒ") === -1 && html.indexOf("Γ") === -1)
  ) {
    return html;
  }

  return html.replace(LEGACY_ICON_PATTERN, (match) => {
    const alt = LEGACY_ICON_ALTS[match] || "Interface icon";
    return renderLegacyIcon(match, alt, 20, "inline-icon") || match;
  });
}

function setInnerHtml(target, html) {
  if (!target) return;
  target.innerHTML = replaceLegacyIcons(html);
}

function getContrastColor(hexColor, lightFallback = "#ffffff", darkFallback = "#111827") {
  if (!hexColor || typeof hexColor !== "string") {
    return lightFallback;
  }

  const raw = hexColor.trim().replace(/^#/, "");
  if (!raw) {
    return lightFallback;
  }

  const expanded = raw.length === 3
    ? raw.split("").map((c) => c + c).join("")
    : raw.slice(0, 6);

  if (expanded.length !== 6) {
    return lightFallback;
  }

  const r = parseInt(expanded.slice(0, 2), 16);
  const g = parseInt(expanded.slice(2, 4), 16);
  const b = parseInt(expanded.slice(4, 6), 16);

  if (Number.isNaN(r) || Number.isNaN(g) || Number.isNaN(b)) {
    return lightFallback;
  }

  const yiq = (r * 299 + g * 587 + b * 114) / 1000;
  return yiq >= 140 ? darkFallback : lightFallback;
}

function hexToRgba(hexColor, alpha = 1) {
  if (!hexColor || typeof hexColor !== "string") {
    return `rgba(30, 64, 175, ${Math.min(Math.max(alpha, 0), 1)})`;
  }

  const sanitized = hexColor.trim().replace(/^#/, "");
  if (!sanitized) {
    return `rgba(30, 64, 175, ${Math.min(Math.max(alpha, 0), 1)})`;
  }

  const normalized = sanitized.length === 3
    ? sanitized.split("").map((c) => c + c).join("")
    : sanitized.slice(0, 6);

  if (normalized.length !== 6) {
    return `rgba(30, 64, 175, ${Math.min(Math.max(alpha, 0), 1)})`;
  }

  const r = parseInt(normalized.slice(0, 2), 16);
  const g = parseInt(normalized.slice(2, 4), 16);
  const b = parseInt(normalized.slice(4, 6), 16);

  if ([r, g, b].some((channel) => Number.isNaN(channel))) {
    return `rgba(30, 64, 175, ${Math.min(Math.max(alpha, 0), 1)})`;
  }

  const clampedAlpha = Math.min(Math.max(alpha, 0), 1);
  return `rgba(${r}, ${g}, ${b}, ${clampedAlpha})`;
}
